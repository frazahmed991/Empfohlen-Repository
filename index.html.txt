        // Funktion zum Abrufen von URL-Parametern
        function getUrlParameter(name) {
            name = name.replace(/[\[]/, '\\[').replace(/[\]]/, '\\]');
            var regex = new RegExp('[\\?&]' + name + '=([^&#]*)');
            var results = regex.exec(location.search);
            return results === null ? '' : decodeURIComponent(results[1].replace(/\+/g, ' '));
        }

        // Download-Button mit korrektem Tracking einrichten
        function setupDownloadButton() {
            const baseOfferUrl = 'https://rewardorbit.com/r/eyJ0IjoiZW1wZm9obGVuIiwidSI6IjY5MDI1ZDEwOTc2Yjg0NGQ3Y2ZiODljMiIsInRzIjoxNzYzMzIxNDMwNTAxLCJwcm92aWRlciI6InRlc3RlcnVwIiwiZHVhbFN5c3RlbSI6dHJ1ZX0';
            const sub2Param = getUrlParameter('sub2');
            
            let finalUrl = baseOfferUrl;
            if (sub2Param) {
                finalUrl += '?sub2=' + encodeURIComponent(sub2Param);
            }
            
            const downloadBtn = document.getElementById('downloadBtn');
            downloadBtn.addEventListener('click', function() {
                // Konversion verfolgen
                console.log('Download-Button geklickt - Diese Konversion verfolgen!', 'sub2:', sub2Param);
                window.open(finalUrl, '_blank');
            });
        }

        // Einnahmen-Zähler animieren
        function animateCounter() {
            const counter = document.querySelector('.earnings-amount');
            const target = 100;
            let current = 0;
            const increment = target / 50;
            
            const timer = setInterval(() => {
                current += increment;
                if (current >= target) {
                    current = target;
                    clearInterval(timer);
                }
                counter.textContent = `${Math.floor(current)}€+`;
            }, 30);
        }

        // Alles nach dem Laden der Seite starten
        window.addEventListener('load', () => {
            setupDownloadButton();
